Insertion flow:

- validate file(
    seek file(SEEK_END - 0x4)
    read addresses(-0x4, -0x3, -0x2, -0x1) as char(s)
  )
- fill footer size(
    seek file(SEEK_END - 0x8)
    read addresses(-0x8, -0x7, -0x6, -0x5) as int
  )

- seek file(SEEK_END - 0x8 - footer size)
- reconstruct schema(
    read addresses(-0x8 - footer size to - schema size) as schema_t
    read addresses(- schema size to SEEK_END - 0x8) as field_desc_t(s)
  )




Pager insertion flow:

- validate file(

    read addresses(-0x4, -0x3, -0x2, -0x1) as char(s)
